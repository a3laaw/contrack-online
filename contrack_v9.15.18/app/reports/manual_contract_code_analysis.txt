class DatabaseError(Exception):
    """خطأ عام في قاعدة البيانات برسالة عربية واضحة."""


class _SharedConnection:
    """اتصال SQLite مُعاد الاستخدام لكل خيط."""

    __slots__ = ("_raw", "_depth", "_failed")

    def __init__(self, raw: sqlite3.Connection):
        self._raw = raw
        self._depth = 0
        self._failed = False

    def __getattr__(self, name):
        return getattr(self._raw, name)

    def __enter__(self):
        self._depth += 1
        return self

    def __exit__(self, exc_type, exc, tb):
        self._depth = max(0, self._depth - 1)
        if self._depth == 0:
            # الفشل يُحسم عند الخارجية فقط: استثناء غير مُمسَك = rollback
            # كامل (ذرية)، وخطأ ممسوك ومُعالَج داخليًا (إعادة ترقيم، تجاهل
            # اختياري) = commit طبيعي كما كان قبل التداخل.
            try:
                if exc_type is not None:
                    self._raw.rollback()
                else:
                    self._raw.commit()
            finally:
                self._failed = False
        elif exc_type is not None:
            self._failed = True
        return False

    def close(self):
        """الإغلاق الفعلي يتم عبر close_connections فقط."""
        return None

    def _real_close(self):
        try:
            self._raw.close()
        except Exception:
            pass


_conn_store = threading.local()


def get_connection():
    """الحصول على اتصال SQLite مع إعدادات محسّنة للشبكات."""
    os.makedirs(DATA_DIR, exist_ok=True)
    
    shared = getattr(_conn_store, "conn", None)
    path = getattr(_conn_store, "path", None)
    
    if shared is not None and path == DB_PATH:
        if getattr(shared, "_depth", 0) > 0:
            return shared  # داخل معاملة خارجية — متحقق منه ولا حاجة لرحلة شبكة
        # V2.48: التحقق مرة/ثانية فقط بدل كل استدعاء — كان SELECT 1
        # يمثل ~40% من كل الاستعلامات (207 في صفحة العملاء وحدها).
        now = time.monotonic()
        if now - getattr(_conn_store, "checked_at", 0.0) < 1.0:
            return shared
        try:                       # التحقق من صلاحية الاتصال المخزَّن
            shared._raw.execute("SELECT 1")
            _conn_store.checked_at = time.monotonic()
            return shared
        except Exception:
            shared._real_close()
            shared = None
    elif shared is not None:       # تغيّر مسار القاعدة (مجلد شبكة/اختبار)
        shared._real_close()
        _conn_store.checked_at = 0.0
    
    # timeout: عند عمل عدة أجهزة على مجلد شبكة، ينتظر تحرر القفل
    # بدل رسالة database is locked الفورية
    raw = sqlite3.connect(DB_PATH, timeout=30.0)
    raw.row_factory = sqlite3.Row

    # إعدادات محسّنة
    raw.execute("PRAGMA foreign_keys = ON;")
    raw.execute("PRAGMA busy_timeout = 30000;")      # 30 ثانية انتظار
    raw.execute("PRAGMA cache_size=-64000;")          # 64 ميجابايت — قراءات أقل عبر الشبكة
    raw.execute("PRAGMA temp_store=MEMORY;")         # تخزين مؤقت في الذاكرة
    # وضع WAL ممتاز محليًا، لكنه غير موثوق على مجلدات SMB/Windows Server.
    # في وضع الشبكة: DELETE + NORMAL — القاعدة تبقى متسقة دائمًا، وأقصى
    # خسارة عند انقطاع الكهرباء لحظة الحفظ هي آخر معاملة فقط (تُعاد).
    # أما FULL فكانت تُبطئ كل حفظ لثوانٍ عبر الشبكة (flush كامل لكل commit).
    try:
        _on_network = bool(os.path.exists(_NET_CONFIG))
        if _on_network:
            raw.execute("PRAGMA journal_mode=DELETE;")
            raw.execute("PRAGMA synchronous=NORMAL;")
            raw.execute("PRAGMA locking_mode=NORMAL;")
        else:
            raw.execute("PRAGMA journal_mode=WAL;")
            raw.execute("PRAGMA synchronous=NORMAL;")
    except Exception:
        raw.execute("PRAGMA synchronous=NORMAL;")
    
    shared = _SharedConnection(raw)
    _conn_store.conn = shared
    _conn_store.path = DB_PATH
    _conn_store.checked_at = time.monotonic()
    return shared


def atomic(fn):
    """معاملة واحدة للعملية كاملة: commit واحد بدل عشرة (تسريع الشبكة).

    يعتمد على عدّاد التداخل في _SharedConnection: كل with داخلية
    متداخلة لا تحفظ حتى خروج الخارجية. الفشل غير الممسوك = rollback
    كامل (أصح من الحفظ الجزئي السابق).
    """
    import functools

    @functools.wraps(fn)
    def wrapper(*args, **kwargs):
        with get_connection():
            return fn(*args, **kwargs)

    return wrapper


def close_connections() -> None:
    """إغلاق الاتصال المخزَّن — قبل استبدال ملف القاعدة (استعادة نسخة)."""
    shared = getattr(_conn_store, "conn", None)
    if shared is not None:
        shared._real_close()
    _conn_store.conn = None
    _conn_store.path = None
    _conn_store.checked_at = 0.0
